function parseBLE(d) { let i = 1; let status = 0; while (i < d.length) { // Skip the BLE-MIDI timestamp byte if (d[i] & 0x80) { i++; } if (i >= d.length) break; // Read MIDI status byte, if present if (d[i] & 0x80) { status = d[i++]; } if (!status) break; const type = status & 0xF0; if (type === 0xF0) { status = 0; continue; } const count = (type === 0xC0 || type === 0xD0) ? 1 : 2; if (i + count > d.length) break; const d1 = d[i++]; const d2 = count === 2 ? d[i++] : 0; onMidi(status, d1, d2); } }
